function sanitize(value,key=''){
 if(/headers|cookies|postData|responseBody|authorization|access.token|refresh.token/i.test(key))return undefined;
 if(typeof value==='string'&&/url|src|href/i.test(key)){try{const u=new URL(value);return u.protocol==='http:'||u.protocol==='https:'?u.origin+u.pathname:value;}catch{return value;}}
 if(Array.isArray(value))return value.map(x=>sanitize(x)).filter(x=>x!==undefined);
 if(value&&typeof value==='object')return Object.fromEntries(Object.entries(value).map(([k,v])=>[k,sanitize(v,k)]).filter(([,v])=>v!==undefined));return value;
}
